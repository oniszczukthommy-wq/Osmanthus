# Authoring / scoring add-ons that the public source imports but never declared.
#
# Install into the SAME .venv after requirements-core.txt:
#   python -m venv .venv
#   python -m pip install -r music-workbench/requirements-core.txt
#   python -m pip install -r music-workbench/requirements-authoring.txt
#
# Why this file exists
#   * mido            - arrangement_authoring / reorchestration / piano_arrangement
#                       write MIDI before FluidSynth renders it.
#   * music21         - arrangement_authoring imports `from music21 import harmony`
#                       for chord spelling; without it the module fails at import.
#   * onnxruntime     - vocal-score-20261024/game_onnx.py runs the GAME ONNX models
#                       (CPUExecutionProvider only).
#
# numpy must stay on the 1.x line. Demucs 4.0.1 and torch 2.5.1+cpu are built
# against it, and music21 happily upgrades to numpy 2.x if it is not pinned.
# Breaking that pin breaks stem separation and the torch audio path.
numpy==1.26.4

# MIDI writing for arrangement authoring and re-orchestration.
mido==1.3.3

# Chord spelling for the piano/guitar authoring core.
music21==10.5.2

# CPU inference for GAME-1.0.3 singing note extraction.
onnxruntime==1.31.1
